    <!-- ٦. SCRIPTS (سکرێپتی کارکردنی سڕینەوە و لۆدینگ) -->
    <script>
      const resetBtn = document.getElementById('resetBtn');
      const confirmModal = document.getElementById('confirmModal');
      const confirmCancelBtn = document.getElementById('confirmCancelBtn');
      const confirmOkBtn = document.getElementById('confirmOkBtn');
      const loadingOverlay = document.getElementById('loadingOverlay');

      // ١. کاتێک کلیک لە دوگمەی سڕینەوە دەکرێت، نامەکە (مۆداڵ) پیشان بدە
      resetBtn.addEventListener('click', () => {
        confirmModal.classList.remove('hidden');
        confirmModal.classList.add('flex');
      });

      // ٢. داپۆشینی مۆداڵ لەکاتی کلیک لە سەر "پەشیمانبوونەوە"
      confirmCancelBtn.addEventListener('click', () => {
        confirmModal.classList.add('hidden');
        confirmModal.classList.remove('flex');
      });

      // ٣. لەکاتی ڕەزامەندی سڕینەوە، لۆدینگەکەی پیشان بدە
      confirmOkBtn.addEventListener('click', () => {
        // شاردنەوەی مۆداڵی دیالۆگ
        confirmModal.classList.add('hidden');
        confirmModal.classList.remove('flex');

        // نیشاندانی لۆدینگ
        loadingOverlay.classList.remove('hidden');
        loadingOverlay.classList.add('flex');

        setTimeout(() => {
          // شاردنەوەی لۆدینگ پاش سڕینەوە
          loadingOverlay.classList.add('hidden');
          loadingOverlay.classList.remove('flex');
          alert('کرداری سڕینەوە بەسەرکەوتوویی ئەنجامدرا!');
        }, 1200);
      });
    </script>
